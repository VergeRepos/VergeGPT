import time

def flashy_greeting(name: str) -> None:
    """Print a neon-style greeting for VergeGPT users."""
    colors = ["\u001b[38;5;45m", "\u001b[38;5;219m", "\u001b[38;5;190m"]
    reset = "\u001b[0m"
    msg = f"VergeGPT says hi, {name}!"
    for i, ch in enumerate(msg):
        color = colors[i % len(colors)]
        print(color + ch + reset, end="", flush=True)
        time.sleep(0.03)
print()  # newline

if __name__ == "__main__":
    flashy_greeting("Verge")
